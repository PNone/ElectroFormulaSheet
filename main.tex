\section{פורמליזם יחסותי, וקטורים, טנזורים ואינדקסים}
המטריקה במוסכמת \textenglish{Mostly Minus} היא 
\[
\eta_{\mu\nu} = \eta^{\mu\nu} = \text{diag}(1, -1, -1, -1)
\]
\begin{itemize}
    \item \textbf{4-וקטור קונטרה (אינדקס עליון):} $A^\mu = (A^0, \vec{A})$. משתנה תחת $A'^\mu = \Lambda^\mu{}_\nu A^\nu = \frac{\partial x'^\mu}{\partial x^\nu} A^\nu \implies \mathbf{A}' = \mathbf{J} \mathbf{A}$.
    \item \textbf{4-וקטור קו ('' תחתון):} $A_\mu=(A_0, -\vec{A})$. משתנה תחת $A'_\mu =\Lambda_\mu{}^\nu A_\nu = (\Lambda^{-1})^\nu{}_\mu A_\nu = \frac{\partial x^\nu}{\partial x'^\mu} A_\nu \Rightarrow \mathbf{A}' = (\mathbf{J}^{-1})^T \mathbf{A}$.
    \item \textbf{העלאת/הורדת אינדקסים:} $A_\mu = \eta_{\mu\nu} A^\nu$, $A^\mu = \eta^{\mu\nu} A_\nu$.
    \item \textbf{טרנספורמציה של טנזורים (כתיב אינדקסים):} 
    \[
    T'^{\mu\nu} = \Lambda^\mu{}_\alpha \Lambda^\nu{}_\beta T^{\alpha\beta}, \
    T'^\mu{}_\nu = \Lambda^\mu{}_\alpha \Lambda_\nu{}^\beta T^\alpha{}_\beta, 
    T'_{\mu\nu} = \Lambda_\mu{}^\alpha \Lambda_\nu{}^\beta T_{\alpha\beta}
    \]
    \item \textbf{טרנספורמציה של טנזורים (כתיב מטריצי):}
    \begin{itemize}
        \item \textbf{עליונים ($T^{\mu\nu}$):} $\mathbf{T}' = \Lambda \mathbf{T} \Lambda^T$
        \item \textbf{תחתונים ($T_{\mu\nu}$):} $\mathbf{T}' = (\Lambda^{-1})^T \mathbf{T} \Lambda^{-1}$
        \item \textbf{מעורבים ($T^\mu{}_\nu$):} $\mathbf{T}' = \Lambda \mathbf{T} \Lambda^{-1}$
    \end{itemize}

\item \textbf{זמן עצמי ($\tau$) והאינטרוואל השמור ($ds$):}
\begin{itemize}
    \item \textbf{קשר לאורך השמור:} $ds^2 = c^2 d\tau^2 = \eta_{\mu\nu} dx^\mu dx^\nu$
    \item \textbf{קשר לזמן במערכת כללית ($t$):} $d\tau = \frac{ds}{c} = \frac{dt}{\gamma}$
    \item \textbf{אינטגרל במסלול:} $\tau = \int \frac{ds}{c} = \int_{t_1}^{t_2} \sqrt{1 - \frac{v^2(t)}{c^2}} \, dt$
\end{itemize}

\item \textbf{ייצוג טנזורים כמטריצות :}
$
T^\mu{}_\nu \quad \leftrightarrow \quad T_{\text{שורה}, \, \text{עמודה}} = T_{\mu, \nu}
$

\end{itemize}

\section{איזומטריה (טרנספורמציית סימטריה)}
טרנספורמציה $x'^\mu = \Lambda^\mu{}_\nu x^\nu$ היא איזומטריה של מרחב מינקובסקי אם היא שומרת על המטריקה:
\[
\eta_{\alpha\beta} \Lambda^\alpha{}_\mu \Lambda^\beta{}_\nu = \eta_{\mu\nu} \quad \iff \quad \Lambda^T \eta \Lambda = \eta
\]

\section{4-וקטורים חשובים}
\begin{itemize}
    \item \textbf{מיקום:} $x^\mu = (ct, \vec{r})$, \quad $x_\mu = (ct, -\vec{r})$
    \item \textbf{נגזרת (\textenglish{4-Gradient}):} $\partial_\mu \equiv \frac{\partial}{\partial x^\mu} = \left(\frac{1}{c}\frac{\partial}{\partial t}, \vec{\nabla}\right)$, \quad $\partial^\mu \equiv \frac{\partial}{\partial x_\mu} = \left(\frac{1}{c}\frac{\partial}{\partial t}, -\vec{\nabla}\right)$
    \item \textbf{מהירות-4:} $U^\mu = \frac{dx^\mu}{d\tau} = \gamma(c, \vec{v})$, כאשר $U^\mu U_\mu = c^2$
    \item \textbf{תנע-אנרגיה 4:} $P^\mu = m U^\mu = \left(\frac{E}{c}, \vec{p}\right) = (\gamma m c, \gamma m \vec{v})$, כאשר $P^\mu P_\mu = m^2 c^2$
    \item \textbf{פוטנציאל-4:} $A^\mu = (\Phi, \vec{A})$, \quad $A_\mu = (\Phi, -\vec{A})$
    \item \textbf{צפיפות זרם-4:} $J^\mu = (\rho c, \vec{J})$, \quad $J_\mu = (\rho c, -\vec{J})$
\end{itemize}

\section{משוואות מקסוול וטנזור השדה $F_{\mu\nu}$}
\begin{itemize}
    \item \textbf{הגדרה:} $F_{\mu\nu} = \partial_\mu A_\nu - \partial_\nu A_\mu$
    \item \textbf{ייצוג מטריצי:}
    {\scriptsize
    \setlength{\arraycolsep}{1.5pt}
    \[
    F_{\mu\nu} = \begin{pmatrix}
    0 & E_x & E_y & E_z \\
    -E_x & 0 & -B_z & B_y \\
    -E_y & B_z & 0 & -B_x \\
    -E_z & -B_y & B_x & 0
    \end{pmatrix}, \;
    F^{\mu\nu} = \begin{pmatrix}
    0 & -E_x & -E_y & -E_z \\
    E_x & 0 & -B_z & B_y \\
    E_y & B_z & 0 & -B_x \\
    E_z & -B_y & B_x & 0
    \end{pmatrix}
    \]
    }
    \item \textbf{חילוץ שדות:} 
    \[
    E^i = F^{i0} = -F^{0i} = F_{0i} = -F_{i0}, \quad
    B^i = \frac{1}{2}\epsilon^{ijk} F^{jk} = \frac{1}{2}\epsilon^{ijk} F_{jk}
    \]
    \item \textbf{משוואות מקסוול (אי-הומוגניות):}
    \[
    \partial_\mu F^{\mu\nu} = \frac{4\pi}{c} J^\nu \iff \vec{\nabla}\cdot\vec{E} = 4\pi\rho, \quad \vec{\nabla}\times\vec{B} - \frac{1}{c}\frac{\partial\vec{E}}{\partial t} = \frac{4\pi}{c}\vec{J}
    \]
    \item \textbf{זהות ביאנקי (משוואות הומוגניות):}
    \[
    \partial_\alpha F_{\beta\gamma} + \partial_\beta F_{\gamma\alpha} + \partial_\gamma F_{\alpha\beta} = 0 \iff \partial_\mu \tilde{F}^{\mu\nu} = 0
    \]
    \[
    \vec{\nabla}\cdot\vec{B} = 0, \quad \vec{\nabla}\times\vec{E} + \frac{1}{c}\frac{\partial\vec{B}}{\partial t} = 0
    \]
\end{itemize}

\textbf{השדה החשמלי והמגנטי באמצעות הפוטנציאלים $(\Phi, \vec{A})$:}
\[
\vec{E} = -\vec{\nabla}\Phi - \frac{1}{c}\frac{\partial \vec{A}}{\partial t}, \quad \vec{B} = \vec{\nabla} \times \vec{A}
\]

\textbf{תנאי שפה וקפיצה על משטח ($\sigma$):} 
\\
$\Delta E_\perp = 4\pi\sigma, \; \Delta \vec{E}_\parallel = 0, \; \Delta\Phi = 0, \; \left.\frac{\partial\Phi_2}{\partial n} - \frac{\partial\Phi_1}{\partial n}\right|_{\text{surface}} = -4\pi\sigma$

\section{הטנזור הדואלי $\tilde{F}^{\mu\nu}$}
הטנזור הדואלי מוגדר ע"י $\tilde{F}^{\mu\nu} = \frac{1}{2}\epsilon^{\mu\nu\alpha\beta}F_{\alpha\beta}$. הוא מתקבל מהחלפת השדות: $\vec{E} \to \vec{B}$ ו-$\vec{B} \to -\vec{E}$.
משמש לכתיבת המשוואות ההומוגניות בקיצור: $\partial_\mu \tilde{F}^{\mu\nu} = 0$.

\section{טנזור מאמץ-אנרגיה $T_{\mu\nu}$}
\begin{itemize}
    \item \textbf{גדלים אלקטרומגנטיים (CGS):}
    \begin{itemize}
        \item צפיפות אנרגיה: $u = \frac{1}{8\pi}(E^2+B^2)$
        \item שטף אנרגיה (וקטור פוינטינג): $\vec{S} = \frac{c}{4\pi}(\vec{E}\times\vec{B})$
        \item צפיפות תנע: $\vec{p}_{\text{EM}} = \frac{1}{c^2}\vec{S} = \frac{1}{4\pi c}(\vec{E}\times\vec{B})$
        \item צפיפות תנע זוויתי: $\vec{\mathcal{L}}_{\text{EM}} = \vec{r} \times \vec{p}_{\text{EM}} = \frac{1}{c^2}(\vec{r} \times \vec{S})$
        \item טנזור המאמצים של מקסוול: $\sigma_{ij} = \frac{1}{4\pi}\left(E_i E_j + B_i B_j - \frac{1}{2}\delta_{ij}(E^2+B^2)\right)$
    \end{itemize}

    \item \textbf{הצגה קונטרא-וואריאנטית ($T^{\mu\nu}$):}
    \[
    T^{\mu\nu} = -\frac{1}{4\pi c} \left( F^\mu{}_\alpha F^{\nu\alpha} - \frac{1}{4}\eta^{\mu\nu}F_{\alpha\beta}F^{\alpha\beta} \right) 
    = \begin{pmatrix} 
    \frac{u}{c} & \vec{p}_{\text{EM}}^T \\
    \frac{1}{c^2}\vec{S} & -\frac{1}{c}\overleftrightarrow{\sigma}
    \end{pmatrix}
    \]

    \item \textbf{הצגה קו-וואריאנטית ($T_{\mu\nu}$):}
    \[
    T_{\mu\nu} = -\frac{1}{4\pi c} \left( F_{\mu\alpha}F_\nu{}^\alpha - \frac{1}{4}\eta_{\mu\nu}F_{\alpha\beta}F^{\alpha\beta} \right)
    = \begin{pmatrix} 
    \frac{u}{c} & -\vec{p}_{\text{EM}}^T \\
    -\frac{1}{c^2}\vec{S} & -\frac{1}{c}\overleftrightarrow{\sigma}
    \end{pmatrix}
    \]

    \item \textbf{הצגה מעורבת ($T^\mu{}_\nu$):}
    \[
    T^\mu{}_\nu = -\frac{1}{4\pi c} \left( F^\mu{}_\alpha F_\nu{}^\alpha - \frac{1}{4}\delta^\mu_\nu F_{\alpha\beta}F^{\alpha\beta} \right) 
    = \begin{pmatrix} 
    \frac{u}{c} & -\vec{p}_{\text{EM}}^T \\
    \frac{1}{c^2}\vec{S} & \frac{1}{c}\overleftrightarrow{\sigma}
    \end{pmatrix}
    \]

    \item \textbf{הצגה מעורבת ($T_\mu{}^\nu$):}
    \[
    T_\mu{}^\nu = -\frac{1}{4\pi c} \left( F_{\mu\alpha}F^{\nu\alpha} - \frac{1}{4}\delta_\mu^\nu F_{\alpha\beta}F^{\alpha\beta} \right) 
    = \begin{pmatrix} 
    \frac{u}{c} & \vec{p}_{\text{EM}}^T \\
    -\frac{1}{c^2}\vec{S} & \frac{1}{c}\overleftrightarrow{\sigma}
    \end{pmatrix}
    \]

    \item \textbf{דיברגנץ' טנזור המאמץ-אנרגיה (שימור תנע-אנרגיה וצפיפות 4-כוח לורנץ):}
    \[
    \partial_\mu T^\mu{}_\nu = -\frac{1}{c^2} F^\mu{}_\nu J_\mu = -f_\nu
    \]
    \begin{itemize}
        \item \textbf{רכיב הזמן ($\nu=0$ - שימור אנרגיה):} $\frac{\partial u}{\partial t} + \vec{\nabla} \cdot \vec{S} = -\vec{J} \cdot \vec{E}$
        \item \textbf{רכיבים מרחביים ($\nu=i$ - שימור תנע):} $\frac{\partial \vec{p}_{\text{EM}}}{\partial t} - \vec{\nabla} \cdot \overleftrightarrow{\sigma} = -\left( \rho \vec{E} + \frac{1}{c} \vec{J} \times \vec{B} \right)$
        \item \textbf{מערכת סגורה (ללא מטענים/זרמים, $J_\mu=0$):} $\partial_\mu T^\mu{}_\nu = 0$
    \end{itemize}
    \item \textbf{עקבת הטנזור (Trace):} $\text{Tr}(T) = T^\mu{}_\mu = 0$.
\end{itemize}

\section{סוגי כיול (\textenglish{Gauge Transformations})}
טרנספורמציית כיול: $A_\mu \to A_\mu + \partial_\mu \Lambda$.
\begin{itemize}
    \item \textbf{כיול לורנץ (\textenglish{Lorenz Gauge}):} תנאי: 
    \[
    \partial_\mu A^\mu = 0 \iff \frac{1}{c}\frac{\partial \Phi}{\partial t} + \vec{\nabla}\cdot\vec{A} = 0
    \]
    מתאים לבעיות יחסותיות וגלים אלקטרומגנטיים. מעביר את משוואות מקסוול לצורה:
    \[
    \square A^\mu = \frac{4\pi}{c}J^\mu \quad \left(\square = \partial_\alpha \partial^\alpha = \frac{1}{c^2}\frac{\partial^2}{\partial t^2} - \nabla^2\right)
    \]
    \item \textbf{כיול קולון (\textenglish{Coulomb Gauge}):} תנאי: $\vec{\nabla}\cdot\vec{A} = 0$.\\
    מתאים לאלקטרוסטטיקה, מגנטוסטטיקה ולקוונטיזציה של השדה. נותן:
    \[
    \nabla^2 \Phi = -4\pi\rho \implies \Phi(\vec{r}, t) = \int \frac{\rho(\vec{r}', t)}{|\vec{r}-\vec{r}'|} d^3 r'
    \]
\end{itemize}

\section{הערך האינטגרלי (הפעולה) ומשוואות שדה}

\begin{itemize}
    \item \textbf{הפעולה הא"מ ($S$):}
    $
    S = \int \mathcal{L}_{\text{EM}} \, d^4x = \int \left( \mathcal{L}_{\text{field}} + \mathcal{L}_{\text{int}} \right) d^4x
    $
    כאשר $d^4x = c \, dt \, d^3x$.

    \item \textbf{צפיפות הלגראנז'יאן :}
    $
    \mathcal{L}_{\text{EM}} = -\frac{1}{16\pi c} F_{\mu\nu}F^{\mu\nu} - \frac{1}{c^2} J_\mu A^\mu
    $
    \begin{itemize}
        \item \textbf{איבר השדה החופשי:} $\mathcal{L}_{\text{field}} = -\frac{1}{16\pi c} F_{\mu\nu}F^{\mu\nu} = \frac{1}{8\pi c} \left( E^2 - B^2 \right)$
        \item \textbf{איבר האינטראקציה:} $\mathcal{L}_{\text{int}} = -\frac{1}{c^2} J_\mu A^\mu = -\frac{1}{c} \rho \Phi + \frac{1}{c^2} \vec{J} \cdot \vec{A}$
    \end{itemize}

    \item \textbf{משוואות \textenglish{EL} לשדה ($A_\nu$):}
    $
    \partial_\mu \left( \frac{\partial \mathcal{L}_{\text{EM}}}{\partial (\partial_\mu A_\nu)} \right) - \frac{\partial \mathcal{L}_{\text{EM}}}{\partial A_\nu} = 0
    $

    \item \textbf{משוואת \textenglish{EL} לשדה סקלרי ($\phi$):}
$
\partial_\mu \left( \frac{\partial \mathcal{L}}{\partial (\partial_\mu \phi)} \right) - \frac{\partial \mathcal{L}}{\partial \phi} = 0
$

    \item \textbf{נגזרות אינטגרנד הוואריאציה:}
    \[
    \frac{\partial \mathcal{L}_{\text{EM}}}{\partial (\partial_\mu A_\nu)} = -\frac{1}{4\pi c} F^{\mu\nu}, \quad \frac{\partial \mathcal{L}_{\text{EM}}}{\partial A_\nu} = -\frac{1}{c^2} J^\nu
    \]

    \item \textbf{הצבה וואריאציה ($\delta S = 0$) $\Leftarrow$ מ' מקסוול לא-הומוגניות:}
    $
    \partial_\mu F^{\mu\nu} = \frac{4\pi}{c} J^\nu
    $
\end{itemize}

\section{פונקציות גרין ומטעני דמות}
\begin{itemize}
    \item \textbf{הגדרת $r_<$ ו-$r_>$:} $r_< = \min(r, r')$, \quad $r_> = \max(r, r')$.

    \item \textbf{מישור אינסופי מוארק ($z=0$):} 
    מטען מקורי $q$ ממוקם ב-$\vec{r}' = (x', y', z')$ (כאשר $z' > 0$). מטען דמות $q' = -q$ ב-$\vec{r}_{\text{img}} = (x', y', -z')$ ($z' < 0$):
    $
    G(\vec{r},\vec{r}') = \frac{1}{|\vec{r}-\vec{r}'|} - \frac{1}{|\vec{r}-\vec{r}_{\text{img}}|}
    $

    \item \textbf{ספירה מוארקת ברדיוס $R$ (תנאי דיריכלה):}
    מטען $q$ ממוקם ב-$\vec{r}'$. מטען דמות $q' = -q \frac{R}{r'}$ ממוקם ב-$\vec{r}_{\text{img}} = \frac{R^2}{r'^2}\vec{r}'$.
    \[
    G(\vec{r},\vec{r}') = \frac{1}{|\vec{r}-\vec{r}'|} - \frac{R/r'}{\left|\vec{r} - \frac{R^2}{r'^2}\vec{r}'\right|}
    \]
    כאשר $\cos\gamma = \hat{r}\cdot\hat{r}' = \cos\theta\cos\theta' + \sin\theta\sin\theta'\cos(\phi-\phi')$:
    \[
    G(\vec{r},\vec{r}') = \frac{1}{\sqrt{r^2 + r'^2 - 2rr'\cos\gamma}} - \frac{1}{\sqrt{\frac{r^2 r'^2}{R^2} + R^2 - 2rr'\cos\gamma}}
    \]

    \item \textbf{פיתוח בכדוריות (פונקציית גרין דיריכלה לספירה ברדיוס $R$):}
    \begin{itemize}
        \item \textbf{בתוך הספירה ($r, r' < R$):}
        {\small
        $
        G_D(\vec{r},\vec{r}') = \sum_{\ell=0}^\infty \sum_{m=-\ell}^\ell \frac{4\pi}{2\ell+1} \Big[ \frac{r_<^\ell}{r_>^{\ell+1}} - \frac{(r \, r')^\ell}{R^{2\ell+1}} \Big] Y_{\ell,m}^*(\theta',\phi') Y_{\ell,m}(\theta,\phi)
        $
        }
        \item \textbf{מחוץ לספירה ($r, r' > R$):}
        {\small
        $
        G_D(\vec{r},\vec{r}') = \sum_{\ell=0}^\infty \sum_{m=-\ell}^\ell \frac{4\pi}{2\ell+1} \Big[ \frac{r_<^\ell}{r_>^{\ell+1}} - \frac{R^{2\ell+1}}{(r \, r')^{\ell+1}} \Big] Y_{\ell,m}^*(\theta',\phi') Y_{\ell,m}(\theta,\phi)
        $
        }
    \end{itemize}

    \item \textbf{משפט התוספת במרחב החופשי:}
    \[
    \frac{1}{|\vec{r} - \vec{r}'|} = \sum_{l=0}^{\infty} \frac{r_<^l}{r_>^{l+1}} P_l(\cos\gamma)
    \]
    \[
    P_l(\cos\gamma) = \frac{4\pi}{2l+1} \sum_{m=-l}^{l} Y_{lm}^*(\theta', \phi') Y_{lm}(\theta, \phi)
    \]
\end{itemize}

\section{פוטנציאל בסימטריה כדורית / פיתוח מולטיפולי}
פוטנציאל מפיזור מטען $\rho(\vec{r}')$:
\[
\Phi(\vec{r}) = 4\pi \sum_{\ell=0}^\infty \sum_{m=-\ell}^\ell \frac{q_{\ell,m}}{2\ell+1} \frac{Y_{\ell,m}(\theta,\phi)}{r^{\ell+1}} \quad (r > r_{\max} \text{עבור })
\]
כאשר המומנטים המולטיפוליים הם:
\[
q_{\ell,m} = \int \rho(\vec{r}') r'^\ell Y_{\ell,m}^*(\theta',\phi') d^3r', \quad q_{\ell,-m} = (-1)^m q_{\ell,m}^*
\]
\textbf{טנזור הקוודרופול:} $Q_{ij} = \int \rho(\vec{r}') (3x'_i x'_j - r'^2 \delta_{ij}) d^3r' \implies \text{Tr}(Q) = \sum_{i} Q_{ii} = 0$.

\vspace{2pt}
\textbf{פוטנציאל של קליפה כדורית ברדיוס $R$:}
\begin{itemize}
    \item \textbf{כללי (תלוי ב-$\theta,\phi$, במונחי הרמוניות כדוריות $Y_{\ell,m}$):}
    \[
    \Phi_{\text{in}} = \sum_{\ell,m} A_{\ell,m} r^\ell Y_{\ell,m}(\theta,\phi), \quad \Phi_{\text{out}} = \sum_{\ell,m} \frac{B_{\ell,m}}{r^{\ell+1}} Y_{\ell,m}(\theta,\phi)
    \]
    \item \textbf{סימטריה אזימותלית ($m=0$):} הפירוק מצטמצם לפולינומי לז'נדר $P_\ell(\cos\theta)$ עם מקדמים $A_\ell, B_\ell$.

    \item \textbf{תכונות אורתוגונליות ונירמול:}
    \begin{itemize}
        \item \textbf{פולינומי לז'נדר ($P_\ell(x)$):} 
        {\small $
        \int_{-1}^{1} P_\ell(x) P_{\ell'}(x) \, dx = \int_{0}^{\pi} P_\ell(\cos\theta) P_{\ell'}(\cos\theta) \sin\theta \, d\theta = \frac{2}{2\ell+1} \delta_{\ell\ell'}
        $}
        \item \textbf{פונקציות לז'נדר המיוחסות ($P_\ell^m(x)$):}
        \[
        \int_{-1}^{1} P_\ell^m(x) P_{\ell'}^m(x) \, dx = \frac{2}{2\ell+1} \frac{(\ell+m)!}{(\ell-m)!} \delta_{\ell\ell'}
        \]
        \item \textbf{הרמוניות כדוריות ($Y_{\ell,m}(\theta,\phi)$):}
        \[
        \int_{0}^{2\pi} d\phi \int_{0}^{\pi} \sin\theta \, d\theta \, Y_{\ell,m}^*(\theta,\phi) Y_{\ell',m'}(\theta,\phi) = \delta_{\ell\ell'} \delta_{mm'}
        \]
    \end{itemize}
\end{itemize}

\section{וקטורי יחידה במערכות קואורדינטות}
\begin{itemize}
    \item \textbf{כדוריות לקרטזיות:}
    $\hat{r} = \sin\theta\cos\phi\hat{x} + \sin\theta\sin\phi\hat{y} + \cos\theta\hat{z}$, \quad
    $\hat{\theta} = \cos\theta\cos\phi\hat{x} + \cos\theta\sin\phi\hat{y} - \sin\theta\hat{z}$, \quad
    $\hat{\phi} = -\sin\phi\hat{x} + \cos\phi\hat{y}$
    \item \textbf{קרטזיות לכדוריות:}
    $\hat{x} = \sin\theta\cos\phi\hat{r} + \cos\theta\cos\phi\hat{\theta} - \sin\phi\hat{\phi}$, \quad
    $\hat{y} = \sin\theta\sin\phi\hat{r} + \cos\theta\sin\phi\hat{\theta} + \cos\phi\hat{\phi}$, \quad
    $\hat{z} = \cos\theta\hat{r} - \sin\theta\hat{\theta}$
    \item \textbf{גליליות לקרטזיות:}
    $\hat{r} = \cos\phi\hat{x} + \sin\phi\hat{y}$, \quad
    $\hat{\phi} = -\sin\phi\hat{x} + \cos\phi\hat{y}$, \quad
    $\hat{z} = \hat{z}$
    \item \textbf{קרטזיות לגליליות:}
    $\hat{x} = \cos\phi\hat{r} - \sin\phi\hat{\phi}$, \quad
    $\hat{y} = \sin\phi\hat{r} + \cos\phi\hat{\phi}$, \quad
    $\hat{z} = \hat{z}$
\end{itemize}

\section{מכפלה וקטורית (\textenglish{Cross Product}) של וקטורי יחידה}
\begin{itemize}
    \item \textbf{כדוריות ($\hat{r}, \hat{\theta}, \hat{\phi}$):}
    $\hat{r} \times \hat{\theta} = \hat{\phi}, \quad \hat{\theta} \times \hat{\phi} = \hat{r}, \quad \hat{\phi} \times \hat{r} = \hat{\theta}$
    \item \textbf{גליליות ($\hat{r}, \hat{\phi}, \hat{z}$):}
    $\hat{r} \times \hat{\phi} = \hat{z}, \quad \hat{\phi} \times \hat{z} = \hat{r}, \quad \hat{z} \times \hat{r} = \hat{\phi}$
\end{itemize}
