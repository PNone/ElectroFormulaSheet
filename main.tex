\section{פורמליזם יחסותי, וקטורים, טנזורים ואינדקסים}
המטריקה במוסכמת \textenglish{Mostly Minus} היא 
\[
\eta_{\mu\nu} = \eta^{\mu\nu} = \text{diag}(1, -1, -1, -1)
\]
\begin{itemize}
    \item \textbf{4-וקטור קונטרא-וואריאנטי (אינדקס עליון):} $A^\mu = (A^0, A^1, A^2, A^3) = (A^0, \vec{A})$. משתנה תחת טרנספורמציה $A'^\mu = \Lambda^\mu{}_\nu A^\nu$. הבסיס מתמרן הפוך.
    \item \textbf{4-וקטור קו-וואריאנטי (אינדקס תחתון):} $A_\mu = (A_0, A_1, A_2, A_3) = (A^0, -\vec{A})$. משתנה תחת טרנספורמציה $A'_\mu = \Lambda_\mu{}^\nu A_\nu = (\Lambda^{-1})^\nu{}_\mu A_\nu$.
    \item \textbf{ייצוג מטריצי של כללי מעבר קואורדינטות:}
    תחת שינוי קואורדינטות $x'^\mu(x)$:
    \[
    A'^\mu = \frac{\partial x'^\mu}{\partial x^\nu} A^\nu \implies \mathbf{A}' = \mathbf{J} \, \mathbf{A} \quad \text{קונטרא:}
    \]
    \[
    A'_\mu = \frac{\partial x^\nu}{\partial x'^\mu} A_\nu \implies \mathbf{A}' = (\mathbf{J}^{-1})^T \mathbf{A} \quad \text{קו:}
    \]
    כאשר $J^\mu{}_\nu = \frac{\partial x'^\mu}{\partial x^\nu}$ היא מטריצת היקוביאן $\mathbf{J}$.
    \item \textbf{העלאת/הורדת אינדקסים:} $A_\mu = \eta_{\mu\nu} A^\nu$, $A^\mu = \eta^{\mu\nu} A_\nu$.
    \item \textbf{טרנספורמציה של טנזורים:} 
    \[
    T'^{\mu\nu} = \Lambda^\mu{}_\alpha \Lambda^\nu{}_\beta T^{\alpha\beta},
    \]
    \[
    T'^\mu{}_\nu = \Lambda^\mu{}_\alpha (\Lambda^{-1})^\beta{}_\nu T^\alpha{}_\beta,
    \]
    \[
    T'_{\mu\nu} = (\Lambda^{-1})^\alpha{}_\mu (\Lambda^{-1})^\beta{}_\nu T_{\alpha\beta}
    \]
    \item \textbf{בכתיב וקטורי/מטריצי:}
\begin{itemize}
    \item \textbf{אינדקסים עליונים ($T^{\mu\nu}$):} 
    $\mathbf{T}' = \Lambda \mathbf{T} \Lambda^T$
    \item \textbf{אינדקסים תחתונים ($T_{\mu\nu}$):} 
    $\mathbf{T}' = (\Lambda^{-1})^T \mathbf{T} \Lambda^{-1}$
    \item \textbf{אינדקסים מעורבים ($T^\mu{}_\nu$):} 
    $\mathbf{T}' = \Lambda \mathbf{T} \Lambda^{-1}$
\end{itemize}
\end{itemize}

\section{איזומטריה (טרנספורמציית סימטריה)}
טרנספורמציה $x'^\mu = \Lambda^\mu{}_\nu x^\nu$ היא איזומטריה של מרחב מינקובסקי אם היא שומרת על המטריקה:
\[
\eta_{\alpha\beta} \Lambda^\alpha{}_\mu \Lambda^\beta{}_\nu = \eta_{\mu\nu} \quad \iff \quad \Lambda^T \eta \Lambda = \eta
\]

\section{4-וקטורים חשובים}
\begin{itemize}
    \item \textbf{מיקום:} $x^\mu = (ct, \vec{r})$, \quad $x_\mu = (ct, -\vec{r})$
    \item \textbf{נגזרת (\textenglish{4-Gradient}):} $\partial_\mu \equiv \frac{\partial}{\partial x^\mu} = \left(\frac{1}{c}\frac{\partial}{\partial t}, \vec{\nabla}\right)$, \quad $\partial^\mu \equiv \frac{\partial}{\partial x_\mu} = \left(\frac{1}{c}\frac{\partial}{\partial t}, -\vec{\nabla}\right)$
    \item \textbf{מהירות-4:} $U^\mu = \frac{dx^\mu}{d\tau} = \gamma(c, \vec{v})$, כאשר $U^\mu U_\mu = c^2$
    \item \textbf{תנע-אנרגיה 4:} $P^\mu = m U^\mu = \left(\frac{E}{c}, \vec{p}\right) = (\gamma m c, \gamma m \vec{v})$, כאשר $P^\mu P_\mu = m^2 c^2$
    \item \textbf{פוטנציאל-4:} $A^\mu = (\Phi, \vec{A})$, \quad $A_\mu = (\Phi, -\vec{A})$
    \item \textbf{צפיפות זרם-4:} $J^\mu = (\rho c, \vec{J})$, \quad $J_\mu = (\rho c, -\vec{J})$
\end{itemize}

\section{משוואות מקסוול וטנזור השדה $F_{\mu\nu}$}
\begin{itemize}
    \item \textbf{הגדרה:} $F_{\mu\nu} = \partial_\mu A_\nu - \partial_\nu A_\mu$
    \item \textbf{ייצוג מטריצי:}
    {\scriptsize
    \setlength{\arraycolsep}{1.5pt}
    \[
    F_{\mu\nu} = \begin{pmatrix}
    0 & E_x & E_y & E_z \\
    -E_x & 0 & -B_z & B_y \\
    -E_y & B_z & 0 & -B_x \\
    -E_z & -B_y & B_x & 0
    \end{pmatrix}, \;
    F^{\mu\nu} = \begin{pmatrix}
    0 & -E_x & -E_y & -E_z \\
    E_x & 0 & -B_z & B_y \\
    E_y & B_z & 0 & -B_x \\
    E_z & -B_y & B_x & 0
    \end{pmatrix}
    \]
    }
    \item \textbf{חילוץ שדות:} 
    \[
    E^i = F^{i0} = -F^{0i} = F_{0i} = -F_{i0},
    B^i = \frac{1}{2}\epsilon^{ijk} F^{jk} = \frac{1}{2}\epsilon^{ijk} F_{jk}
    \]
    \item \textbf{משוואות מקסוול (אי-הומוגניות):}
    \[
    \partial_\mu F^{\mu\nu} = \frac{4\pi}{c} J^\nu \iff \vec{\nabla}\cdot\vec{E} = 4\pi\rho, \quad \vec{\nabla}\times\vec{B} - \frac{1}{c}\frac{\partial\vec{E}}{\partial t} = \frac{4\pi}{c}\vec{J}
    \]
    \item \textbf{זהות ביאנקי (משוואות הומוגניות):}
    \[
    \partial_\alpha F_{\beta\gamma} + \partial_\beta F_{\gamma\alpha} + \partial_\gamma F_{\alpha\beta} = 0 \iff \partial_\mu \tilde{F}^{\mu\nu} = 0
    \]
    \[
    \vec{\nabla}\cdot\vec{B} = 0, \quad \vec{\nabla}\times\vec{E} + \frac{1}{c}\frac{\partial\vec{B}}{\partial t} = 0
    \]
\end{itemize}

\section{הטנזור הדואלי $\tilde{F}^{\mu\nu}$}
הטנזור הדואלי מוגדר ע"י $\tilde{F}^{\mu\nu} = \frac{1}{2}\epsilon^{\mu\nu\alpha\beta}F_{\alpha\beta}$. הוא מתקבל מהחלפת השדות: $\vec{E} \to \vec{B}$ ו-$\vec{B} \to -\vec{E}$.
משמש לכתיבת המשוואות ההומוגניות בקיצור: $\partial_\mu \tilde{F}^{\mu\nu} = 0$.

\section{גודלים אינווריאנטיים של השדה}
\begin{itemize}
    \item $F_{\mu\nu}F^{\mu\nu} = 2(B^2 - E^2)$
    \item $F_{\mu\nu}\tilde{F}^{\mu\nu} = -4(\vec{E}\cdot\vec{B})$
\end{itemize}

\section{טנזור מאמץ-אנרגיה $T_{\mu\nu}$}
\begin{itemize}
    \item \textbf{הגדרה טנזורית:} $T_{\mu\nu} = -\frac{1}{4\pi c} \left( F_{\mu\alpha}F_\nu{}^\alpha - \frac{1}{4}\eta_{\mu\nu}F_{\alpha\beta}F^{\alpha\beta} \right)$
    \item \textbf{רכיבים ומטריצה (אינדקסים תחתונים):}
    \[
    T_{\mu\nu} = \frac{1}{c} \begin{pmatrix} 
    u & -\frac{1}{c}S_x & -\frac{1}{c}S_y & -\frac{1}{c}S_z \\
    -\frac{1}{c}S_x & -\sigma_{xx} & -\sigma_{xy} & -\sigma_{xz} \\
    -\frac{1}{c}S_y & -\sigma_{yx} & -\sigma_{yy} & -\sigma_{yz} \\
    -\frac{1}{c}S_z & -\sigma_{zx} & -\sigma_{zy} & -\sigma_{zz}
    \end{pmatrix}
    \]
    \\
    כאשר $u = \frac{1}{8\pi}(E^2+B^2) $ (צפיפות אנרגיה), \\
    $\vec{S} = \frac{c}{4\pi}(\vec{E}\times\vec{B})$ (וקטור פוינטינג), ו-\\$\sigma_{ij} = \frac{1}{4\pi}\left(E_i E_j + B_i B_j - \frac{1}{2}\delta_{ij}(E^2+B^2)\right)$ הוא טנזור המאמצים של מקסוול.
    \item \textbf{הצגה קונטרא-וואריאנטית ($T^{\mu\nu}$):} 
    \[
    T^{\mu\nu} = \eta^{\mu\alpha}\eta^{\nu\beta} T_{\alpha\beta} = \frac{1}{c} \begin{pmatrix}
    u & \frac{1}{c}S_x & \frac{1}{c}S_y & \frac{1}{c}S_z \\
    \frac{1}{c}S_x & -\sigma_{xx} & -\sigma_{xy} & -\sigma_{xz} \\
    \frac{1}{c}S_y & -\sigma_{yx} & -\sigma_{yy} & -\sigma_{yz} \\
    \frac{1}{c}S_z & -\sigma_{zx} & -\sigma_{zy} & -\sigma_{zz}
    \end{pmatrix}
    \]
    \item \textbf{הצגה מעורבת ($T^\mu{}_\nu$):}
    \[
    T^\mu{}_\nu = T^{\mu\alpha}\eta_{\alpha\nu} = -\frac{1}{4\pi c} \left( F^{\mu\alpha}F_{\nu\alpha} - \frac{1}{4}\delta^\mu_\nu F_{\alpha\beta}F^{\alpha\beta} \right)
    \]
    \[
    T^\mu{}_\nu = \frac{1}{c} \begin{pmatrix}
    u & -\frac{1}{c}S_x & -\frac{1}{c}S_y & -\frac{1}{c}S_z \\
    \frac{1}{c}S_x & \sigma_{xx} & \sigma_{xy} & \sigma_{xz} \\
    \frac{1}{c}S_y & \sigma_{yx} & \sigma_{yy} & \sigma_{yz} \\
    \frac{1}{c}S_z & \sigma_{zx} & \sigma_{zy} & \sigma_{zz}
    \end{pmatrix}
    \]
    \item \textbf{חוק שימור:} 
    \[
    \partial^\mu T_{\mu\nu} = -\frac{1}{c^2} F_{\nu\alpha}J^\alpha \quad \left(\partial_\mu T^{\mu\nu} = -\frac{1}{c^2} F^{\nu\alpha}J_\alpha \text{או } \right)
    \]
\end{itemize}

\section{סוגי כיול (\textenglish{Gauge Transformations})}
טרנספורמציית כיול: $A_\mu \to A_\mu + \partial_\mu \Lambda$.
\begin{itemize}
    \item \textbf{כיול לורנץ (\textenglish{Lorenz Gauge}):} תנאי: 
    \[
    \partial_\mu A^\mu = 0 \iff \frac{1}{c}\frac{\partial \Phi}{\partial t} + \vec{\nabla}\cdot\vec{A} = 0
    \]
    מתאים לבעיות יחסותיות וגלים אלקטרומגנטיים. מעביר את משוואות מקסוול לצורה:
    \[
    \square A^\mu = \frac{4\pi}{c}J^\mu \quad \left(\square = \partial_\alpha \partial^\alpha = \frac{1}{c^2}\frac{\partial^2}{\partial t^2} - \nabla^2 \text{כאשר } \right)
    \]
    \item \textbf{כיול קולון (\textenglish{Coulomb Gauge}):} תנאי: $\vec{\nabla}\cdot\vec{A} = 0$.\\
    מתאים לאלקטרוסטטיקה, מגנטוסטטיקה ולקוונטיזציה של השדה. נותן:
    \[
    \nabla^2 \Phi = -4\pi\rho \implies \Phi(\vec{r}, t) = \int \frac{\rho(\vec{r}', t)}{|\vec{r}-\vec{r}'|} d^3 r'
    \]
\end{itemize}

\section{פעולה ומשוואות אוילר-לגראנז'}
הפעולה היחסותית היא $S = \int \mathcal{L} \, d^4x$, כאשר צפיפות הלגראנז'יאן מתפרקת ל:
\[
\mathcal{L}_{EM} = \mathcal{L}_{\text{field}} + \mathcal{L}_{\text{int}} = -\frac{1}{16\pi} F_{\mu\nu}F^{\mu\nu} - \frac{1}{c}J_\mu A^\mu
\]
משוואות אוילר-לגראנז' עבור השדות:
\[
\frac{\partial \mathcal{L}}{\partial A_\nu} - \partial_\mu \left( \frac{\partial \mathcal{L}}{\partial(\partial_\mu A_\nu)} \right) = 0 \implies \partial_\mu F^{\mu\nu} = \frac{4\pi}{c} J^\nu
\]

\textbf{חילוף סדר נגזרת ווריאציה:} מותר להחליף את סדר הנגזרת והווריאציה במרחב המסלולים:
$
\delta \left( \frac{dx^\mu}{d\lambda} \right) = \frac{d}{d\lambda} (\delta x^\mu)
$

\section{פונקציות גרין ופוטנציאל}
חישוב הפוטנציאל הפואסוני $\nabla^2 \Phi = -4\pi \rho$:
\[
\Phi(\vec{r}) = \int_V \rho(\vec{r}') G(\vec{r}, \vec{r}') d^3r' + \frac{1}{4\pi} \oint_{\partial V} \left( G \vec{\nabla}'\Phi - \Phi \vec{\nabla}'G \right) \cdot d\vec{A}'
\]
\begin{itemize}
    \item \textbf{תנאי דיריכלה ($G_D=0$ על השפה):}
    \[
    \Phi(\vec{r}) = \int_V \rho G_D d^3r' - \frac{1}{4\pi} \oint_{\partial V} \Phi(\vec{r}') \frac{\partial G_D}{\partial n'} dA'
    \]
    \item \textbf{תנאי נוימן ($\frac{\partial G_N}{\partial n'} = \frac{4\pi}{S}$ על השפה):}
    \[
    \Phi(\vec{r}) = \langle\Phi\rangle_S + \int_V \rho G_N d^3r' + \frac{1}{4\pi} \oint_{\partial V} \frac{\partial \Phi}{\partial n'} G_N dA'
    \]
\end{itemize}

\section{פונקציות גרין ומטעני דמות}
\begin{itemize}
    \item \textbf{מישור אינסופי מוארק ($z=0$):} $G(\vec{r},\vec{r}') = \frac{1}{|\vec{r}-\vec{r}'|} - \frac{1}{|\vec{r}-\vec{r}_{\text{img}} |}$ כאשר $\vec{r}_{\text{img}} = (x', y', -z')$. (מטען דמות $-q$ ב-$z'=-d$).
    \item \textbf{ספירה מוארקת ברדיוס $R$:} מטען דמות $q' = -q \frac{R}{d}$ במיקום $r' = \frac{R^2}{d}$.
    \[
    G(\vec{r},\vec{r}') = \frac{1}{|\vec{r}-\vec{r}'|} - \frac{R/r'}{\left|\vec{r} - \frac{R^2}{r'^2}\vec{r}'\right|}
    \]
    \item \textbf{פיתוח בכדוריות:} 
    \[
    \frac{1}{|\vec{r}-\vec{r}'|} = \sum_{\ell=0}^\infty \sum_{m=-\ell}^\ell \frac{4\pi}{2\ell+1} \frac{r_<^\ell}{r_>^{\ell+1}} Y_{\ell,m}^*(\theta',\phi') Y_{\ell,m}(\theta,\phi)
    \]
\end{itemize}

\section{פוטנציאל בסימטריה כדורית / פיתוח מולטיפולי}
פוטנציאל מפיזור מטען $\rho(\vec{r}')$:
\[
\Phi(\vec{r}) = 4\pi \sum_{\ell=0}^\infty \sum_{m=-\ell}^\ell \frac{q_{\ell,m}}{2\ell+1} \frac{Y_{\ell,m}(\theta,\phi)}{r^{\ell+1}} \quad ( r > r_{\max} \text{עבור })
\]
כאשר המומנטים המולטיפוליים הם 
\[
q_{\ell,m} = \int \rho(\vec{r}') r'^\ell Y_{\ell,m}^*(\theta',\phi') d^3r'.
\]
